    \end{column}
    \begin{column}{0.315\textwidth}
      \stat{+11pp}{olens}{\olens{} over the prompt on the 1,122 cases Qwen gets wrong}
    \end{column}
  \end{columns}
  \medskip
  \begin{itemize}
    \item Haiku predicts Qwen3.6-27B's SuperGPQA letter from the prompt, with and without lens reads
    \item \jlens{} and \nla{} beat their placebos by about 3pp; no placebo beats the prompt alone
  \end{itemize}
  \takeaway{The O-lens reads Qwen's answer before Qwen writes it, most of all when Qwen is wrong.}
\end{frame}

\begin{frame}{The question}
  Given only the prompt, predict the letter Qwen3.6-27B will answer.
  \begin{itemize}
    \item 3,000 SuperGPQA questions where Qwen gives one letter in $\geq$ 7 of 8 samples
    \item Investigator: Claude Haiku 5.5, one draw, a single \texttt{submit\_answer} call
    \item Seven conditions: \transcript, and the \olens, \jlens{} and \nla{} reads of the prompt, each with a placebo
      \begin{itemize}
        \item a placebo is another case's reads, presented as this case's
      \end{itemize}
    \item Truth is Qwen's modal letter, not the gold answer
  \end{itemize}
  \takeaway{If the lens only paraphrases the prompt, the placebo should score as well as the lens.}
\end{frame}
